# Libraries the AppImage takes from the host (not bundled). One shell glob per line, matched against
# the soname. Everything else an ELF file in the AppDir needs must resolve inside the AppDir, or the
# build fails (build.sh, "library audit").
#
# glibc: always the host's (the build baseline is glibc 2.35, so hosts need glibc >= 2.35).
ld-linux-x86-64.so.2
libc.so.6
libm.so.6
libdl.so.2
libpthread.so.0
librt.so.1
libresolv.so.2
libutil.so.1
libanl.so.1
libnsl.so.1
libnss_*.so.2
# GPU / GL / display stack: must match the host's drivers and display server.
libGL.so.1
libGLX.so.0
libGLdispatch.so.0
libOpenGL.so.0
libEGL.so.1
libgbm.so.1
libdrm.so.2
libdrm_*.so.*
libX11.so.6
libX11-xcb.so.1
libxcb.so.1
# xcb extensions shared with the host's GL drivers; cairo's own (render, shm) are bundled.
libxcb-dri2.so.0
libxcb-dri3.so.0
libxcb-glx.so.0
libxcb-present.so.0
libxcb-sync.so.1
# (libwayland-client / -cursor / -egl are bundled: GTK links them, and hosts without Wayland lack them;
# GTK 3 draws with cairo, not EGL, so an older bundled copy does not meet the host's GPU drivers.)
# Fonts come from the host, with its own font configuration (a bundled fontconfig could not parse a newer
# host configuration).
libfontconfig.so.1
libfreetype.so.6
# C++ / GCC runtime: the host's is always at least as new as Ubuntu 22.04's, and host GPU drivers need it.
libstdc++.so.6
libgcc_s.so.1
